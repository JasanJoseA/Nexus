async function runPreview(){
  const btn = document.getElementById('run-preview-btn');
  const frameWrap = document.getElementById('browser-frame');
  const loading = document.getElementById('preview-loading');
  const loadingText = document.getElementById('preview-loading-text');
  const host = document.getElementById('preview-frame');
  const urlBar = document.getElementById('browser-url');

  btn.disabled = true;
  btn.textContent = '● Compiling...';
  frameWrap.classList.add('show');
  loading.style.display='flex';
  host.style.display='none';

  const msgs = ['Compiling build...','Bundling components...','Applying design tokens...','Rendering homepage...'];
  let mi=0;
  const cycle = setInterval(()=>{ mi=(mi+1)%msgs.length; loadingText.textContent = msgs[mi]; }, 900);

  try{
    const bundle = buildPreviewBundle();
    const html = bundle.html;
    state.previewHtml = html;

    clearInterval(cycle);
    host.innerHTML = '';
    const iframe = document.getElementById('site-preview-iframe');
    iframe.srcdoc = html;
    loading.style.display='none';
    host.style.display='block';
    const slug = (state.name||'project').toLowerCase().replace(/[^a-z0-9]+/g,'-').replace(/(^-|-$)/g,'');
    urlBar.textContent = `localhost:3000/${slug}`;
    log(`[PREVIEW] live build rendered ✓`, 't-sys');
    btn.textContent = '↺ Rebuild Preview';
    if(document.getElementById('files-panel').classList.contains('show')) buildFilesPanel();
  }catch(err){
    clearInterval(cycle);
    loadingText.textContent = 'Preview failed — click Run Project to retry.';
    log(`[PREVIEW] error: ${(err.message||'request failed')}`, 't-err');
    btn.textContent = '▶ Run Project';
  }
  btn.disabled = false;
}

document.getElementById('run-preview-btn').addEventListener('click', runPreview);

function extractCodeBlock(raw){
  if(!raw) return null;
  const m = raw.match(/```[a-zA-Z]*\n([\s\S]*?)```/);
  return m ? m[1].trim() : null;
}

function buildManifest(){
  const slug = (state.name||'project').toLowerCase().replace(/[^a-z0-9]+/g,'-').replace(/(^-|-$)/g,'') || 'project';
  const feat = selectedFeatures();
  const readme = `# ${state.name || 'Untitled Project'}

${state.type || ''}

## Brief
${document.getElementById('proj-brief').value}

## Capabilities
${feat.length ? feat.map(f=>'- '+f).join('\n') : '- none specified'}

## Generated by
NEXUS Orchestrator — Architect, Designer, Frontend, Backend, Content, and QA agents.
See architecture.md, design-system.md, homepage-copy.md, api-design.md, qa-checklist.md, index.html, styles.css, app.js, and site-data.json in this file set.`;

  const previewBundle = buildPreviewBundle();
  const files = [
    { name:'README.md', content: readme },
    { name:'index.html', content: state.previewHtml || previewBundle.html },
    { name:'styles.css', content: previewBundle.css },
    { name:'app.js', content: `document.addEventListener('DOMContentLoaded', () => {
  const toast = document.getElementById('toast');
  document.querySelectorAll('[data-action]').forEach((button) => {
    button.addEventListener('click', () => {
      const actions = {
        'book-now': 'Your class request is ready.',
        'view-plans': 'Membership plans opened.',
        'learn-more': 'More details are on the way.',
        'select-plan': 'Plan selected. Checkout is ready.'
      };
      const message = actions[button.dataset.action] || 'Action ready.';
      toast.textContent = message;
      toast.classList.add('show');
      clearTimeout(window.__navToastTimer);
      window.__navToastTimer = setTimeout(() => toast.classList.remove('show'), 1800);
    });
  });
});` },
    { name:'site-data.json', content: JSON.stringify(previewBundle.json, null, 2) },
    { name:'architecture.md', content: state.outputs.architect || '// Run a build first.' },
    { name:'design-system.md', content: state.outputs.designer || '// Run a build first.' },
  ];
  const snippet = extractCodeBlock(state.outputs.frontend);
  files.push({ name:'component.jsx', content: snippet || state.outputs.frontend || '// Run a build first.' });
  files.push({ name:'api-design.md', content: state.outputs.backend || '// Run a build first.' });
  files.push({ name:'homepage-copy.md', content: state.outputs.content || '// Run a build first.' });
  files.push({ name:'qa-checklist.md', content: state.outputs.qa || '// Run a build first.' });
  return files;
}

let activeFileIdx = 0;
function buildFilesPanel(){
  const files = buildManifest();
  const tabs = document.getElementById('files-tabs');
  tabs.innerHTML = files.map((f,i)=>``).join('');
  tabs.querySelectorAll('.file-tab').forEach(btn=>{
    btn.addEventListener('click', ()=>{ activeFileIdx = parseInt(btn.dataset.i,10); renderFileTab(files); });
  });
  renderFileTab(files);
}
function renderFileTab(files){
  const f = files[activeFileIdx] || files[0];
  document.querySelectorAll('.file-tab').forEach((t,i)=> t.classList.toggle('active', i===activeFileIdx));
  document.getElementById('files-filename').textContent = f.name;
  document.getElementById('files-content').textContent = f.content;
}
document.getElementById('copy-file-btn').addEventListener('click', ()=>{
  const text = document.getElementById('files-content').textContent;
  const btn = document.getElementById('copy-file-btn');
  const done = ()=>{ btn.textContent='Copied ✓'; setTimeout(()=> btn.textContent='Copy File', 1400); };
  if(navigator.clipboard && navigator.clipboard.writeText){
    navigator.clipboard.writeText(text).then(done).catch(()=>fallbackCopy(text,done));
  } else {
    fallbackCopy(text, done);
  }
});
function fallbackCopy(text, cb){
  const ta = document.createElement('textarea');
  ta.value = text; ta.style.position='fixed'; ta.style.opacity='0';
  document.body.appendChild(ta); ta.focus(); ta.select();
  try{ document.execCommand('copy'); }catch(e){}
  document.body.removeChild(ta);
  cb();
}
document.getElementById('view-files-btn').addEventListener('click', ()=>{
  const panel = document.getElementById('files-panel');
  panel.classList.toggle('show');
  if(panel.classList.contains('show')){ activeFileIdx = 0; buildFilesPanel(); }
});
